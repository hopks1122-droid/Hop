local function getPlayerLevel()

    local ls = LP:FindFirstChild("leaderstats")

    if ls then

        local lv = ls:FindFirstChild("Level") or ls:FindFirstChild("LV") or ls:FindFirstChild("Lv")

        if lv and lv:IsA("IntValue") then return lv.Value end

    end

    return 1

end

local function getQuestByLevel(level)

    for _, q in pairs(questData) do

        if level >= q.min and level < q.max then return q end

    end

    return questData[1]

end

local function isEnemy(obj)

    if not obj or not obj.Parent or not obj:IsA("Model") then return false end

    if obj == LP.Character then return false end

    if PS:GetPlayerFromCharacter(obj) then return false end

    local h = obj:FindFirstChild("Humanoid")

    local r = obj:FindFirstChild("HumanoidRootPart")

    if not h or not r or h.Health <= 0 then return false end

    return true

end

local function isRightEnemy(obj, mobName)

    if not isEnemy(obj) then return false end

    if mobName ~= "" and not obj.Name:lower():find(mobName:lower()) then return false end

    return true

end

local function findAndClickNPC(npcName)

    for _, v in pairs(WS:GetDescendants()) do

        if v:IsA("Model") and v.Name:lower():find(npcName:lower()) then

            local head = v:FindFirstChild("Head")

            if head then

                local c = LP.Character

                if c then

                    local r = c:FindFirstChild("HumanoidRootPart")

                    if r then

                        r.CFrame = CFrame.new(head.Position + Vector3.new(0, 3, 5))

                        wait(0.3)

                        VIM:SendMouseButtonEvent(0, 0, 0, true, game, 0)

                        wait(0.1)

                        VIM:SendMouseButtonEvent(0, 0, 0, false, game, 0)

                        return true

                    end

                end

            end

        end

    end

    return false

end

local function isFruit(obj)

    local n = obj.Name:lower()

    local notFruit = {"wing", "sword", "blade", "gun", "rifle", "accessory", "hat", "cape", "belt", "boot", "shield"}

    for _, kw in pairs(notFruit) do if n:find(kw) then return false end end

    for _, kw in pairs(fruitKeywords) do if n:find(kw) then return true end end

    return false

end
